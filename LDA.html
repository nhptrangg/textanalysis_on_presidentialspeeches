speech <- readRDS("SpeechData.rds")

pivot_date <- as.Date("04-20-2005", format = "%m-%d-%Y")
speech_before <- speech |> filter(date < pivot_date)
speech_after <- speech |> filter(date > pivot_date)

Before pivot point

#convert the raw text to corpus
speech_cor_before <- corpus(speech_before) 
#split the speeches into paragraphs as documents
corp_parag_before <- corpus_reshape(speech_cor_before, to = "paragraphs") 
#convert into token objects and remove the redundancies
toks_before <- quanteda::tokens(
  corp_parag_before,
  remove_punct = TRUE,
  remove_numbers = TRUE,
  remove_symbols = TRUE,
  remove_separators = TRUE
)
toks_before <- tokens_remove(toks_before, pattern = stopwords("english"))

custom_words_to_remove <- c("much", "can", "many", "just", 
                            "must", "Mr", "As", "of", "The", "every")
toks_before <- tokens_remove(toks_before, pattern = custom_words_to_remove)


# Convert tokens to a list of words
toks_list <- as.list(toks_before)  # converts to list of character vectors

# Lemmatize each token list
toks_lemma_list <- lapply(toks_list, lemmatize_words)

# Convert back to tokens
toks_before_lemma <- quanteda::tokens(toks_lemma_list)

#create document feature matrix
#row: documents, columns: topics, value: how frequent the topics appear
dfm_b <- toks_before_lemma |> 
  dfm() |>
  dfm_remove(pattern = stopwords("english"))
#remove words that are less than 5 (reduce the noise)
dfm_b <- dfm_trim(dfm_b, min_docfreq = 5)

#convert into a format that we can feed into LDA
dtm_b <- convert(dfm_b, to = "topicmodels")
set.seed(1)
#LDA is a probabilistic model so the result is always different. Hence, we have to set seeds
#k: specify the number of topics
m_before <- LDA(dtm_b, k = 4, control = list(seed = 123))
m_before
A LDA_VEM topic model with 4 topics.
